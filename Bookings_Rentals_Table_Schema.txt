CREATE TABLE bookings (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    equipment_id UUID NOT NULL,
    renter_id UUID NOT NULL,
    partner_id UUID NOT NULL,
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    pickup_time TIME,
    return_time TIME,
    return_deadline TIME NOT NULL,
    is_early_delivery_requested BOOLEAN NOT NULL,
    is_early_delivery_approved BOOLEAN,
    hotel_dropoff_location VARCHAR(200),
    coordination_notes TEXT,
    theme_park_destination VARCHAR(100),
    special_event VARCHAR(100),
    is_flexible_pickup BOOLEAN NOT NULL,
    late_return_grace_period_in_minutes INTEGER NOT NULL,
    total_days INTEGER NOT NULL,
    daily_rate DECIMAL(10,2) NOT NULL,
    subtotal DECIMAL(10,2) NOT NULL,
    delivery_fee DECIMAL(8,2),
    service_fee DECIMAL(8,2) NOT NULL,
    tax_amount DECIMAL(8,2) NOT NULL,
    total_amount DECIMAL(10,2) NOT NULL,
    security_deposit DECIMAL(10,2) NOT NULL,
    status VARCHAR(50) NOT NULL CHECK (status IN ('pending', 'confirmed', 'cancelled', 'completed', 'in_progress')),
    payment_status VARCHAR(50) NOT NULL CHECK (payment_status IN ('pending', 'paid', 'refunded', 'failed')),
    payment_intent_id VARCHAR(100),
    delivery_method VARCHAR(50) NOT NULL CHECK (delivery_method IN ('pickup', 'delivery', 'hotel_dropoff')),
    delivery_address TEXT,
    pickup_location TEXT,
    special_requests TEXT,
    equipment_condition_start JSON,
    equipment_condition_end JSON,
    is_damage_reported BOOLEAN NOT NULL,
    damage_description TEXT,
    damage_cost DECIMAL(8,2),
    is_late_return BOOLEAN NOT NULL,
    late_fee DECIMAL(8,2),
    confirmed_at TIMESTAMP WITH TIME ZONE,
    cancelled_at TIMESTAMP WITH TIME ZONE,
    cancellation_reason TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (equipment_id) REFERENCES equipments(id),
    FOREIGN KEY (renter_id) REFERENCES renters(id),
    FOREIGN KEY (partner_id) REFERENCES partners(id)
);
