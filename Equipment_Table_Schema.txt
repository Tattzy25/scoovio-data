CREATE TABLE equipments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    partner_id UUID NOT NULL,
    category_id UUID NOT NULL,
    name VARCHAR(255) NOT NULL,
    description TEXT,
    make VARCHAR(255),
    model VARCHAR(255),
    year INT,
    serial_number VARCHAR(255) UNIQUE,
    condition VARCHAR(255),
    daily_rental_rate DECIMAL(10, 2) NOT NULL,
    weekly_rental_rate DECIMAL(10, 2),
    monthly_rental_rate DECIMAL(10, 2),
    min_rental_period_in_days INT,
    max_rental_period_in_days INT,
    availability_status VARCHAR(255),
    address VARCHAR(255),
    city VARCHAR(255),
    state VARCHAR(255),
    zip_code VARCHAR(20),
    country VARCHAR(255),
    delivery_options VARCHAR(255),
    pickup_options VARCHAR(255),
    is_insurance_required BOOLEAN,
    insurance_details TEXT,
    maintenance_history TEXT,
    last_maintenance_date DATE,
    next_maintenance_date DATE,
    purchase_date DATE,
    purchase_price DECIMAL(10, 2),
    depreciation_rate DECIMAL(5, 2),
    current_value DECIMAL(10, 2),
    total_rentals INT DEFAULT 0,
    total_revenue DECIMAL(10, 2) DEFAULT 0.00,
    average_rating DECIMAL(2, 1),
    number_of_reviews INT DEFAULT 0,
    is_featured BOOLEAN DEFAULT FALSE,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (partner_id) REFERENCES partners(id),
    FOREIGN KEY (category_id) REFERENCES equipment_categories(id)
);
