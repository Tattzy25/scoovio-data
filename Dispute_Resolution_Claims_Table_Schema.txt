CREATE TABLE dispute_resolution_claims (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    booking_id UUID NOT NULL,
    renter_id UUID,
    partner_id UUID,
    claim_type VARCHAR(100) NOT NULL,
    description TEXT NOT NULL,
    evidence_urls TEXT[],
    status VARCHAR(50) DEFAULT 'open' NOT NULL CHECK (status IN ('open', 'in_review', 'resolved', 'closed', 'rejected')),
    resolution_date TIMESTAMP WITH TIME ZONE,
    resolved_by_admin_id UUID,
    financial_adjustment DECIMAL(10, 2),
    notes TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (booking_id) REFERENCES bookings(id),
    FOREIGN KEY (renter_id) REFERENCES renters(id),
    FOREIGN KEY (partner_id) REFERENCES partners(id),
    FOREIGN KEY (resolved_by_admin_id) REFERENCES admin_users(id)
);
