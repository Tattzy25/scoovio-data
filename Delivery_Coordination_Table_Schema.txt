CREATE TABLE delivery_coordinations (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    booking_id UUID NOT NULL,
    delivery_address TEXT NOT NULL,
    pickup_address TEXT NOT NULL,
    delivery_date TIMESTAMP WITH TIME ZONE NOT NULL,
    pickup_date TIMESTAMP WITH TIME ZONE NOT NULL,
    status VARCHAR(50) NOT NULL, -- e.g., 'scheduled', 'in_transit', 'delivered', 'failed'
    delivery_driver_id UUID,
    delivery_notes TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (booking_id) REFERENCES bookings(id),
    FOREIGN KEY (delivery_driver_id) REFERENCES users(id) -- Assuming drivers are also users
);
