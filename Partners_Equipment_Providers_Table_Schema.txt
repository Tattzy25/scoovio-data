CREATE TABLE partners (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email VARCHAR(255) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    partner_type VARCHAR(50) NOT NULL,
    business_name VARCHAR(200),
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    phone VARCHAR(20) NOT NULL,
    date_of_birth DATE,
    profile_image_url VARCHAR(500),
    bio TEXT,
    business_address TEXT,
    home_address TEXT,
    city VARCHAR(100) NOT NULL,
    state VARCHAR(50) NOT NULL,
    zip_code VARCHAR(10) NOT NULL,
    country VARCHAR(50) NOT NULL,
    business_license VARCHAR(100),
    ein_tax_id VARCHAR(50),
    ssn_last_four VARCHAR(4),
    tax_filing_method VARCHAR(50) NOT NULL,
    insurance_policy_url VARCHAR(500),
    background_check_id VARCHAR(100),
    background_check_status VARCHAR(50) NOT NULL,
    is_approved BOOLEAN NOT NULL,
    rejection_reason TEXT,
    is_business_verified BOOLEAN NOT NULL,
    average_rating DECIMAL(3,2) NOT NULL,
    total_rentals INTEGER NOT NULL,
    total_listings INTEGER NOT NULL,
    account_status VARCHAR(50) NOT NULL,
    payout_method VARCHAR(50) NOT NULL,
    payout_details JSONB,
    payout_schedule VARCHAR(50) NOT NULL,
    commission_rate DECIMAL(5,2) NOT NULL,
    last_login_at TIMESTAMP WITH TIME ZONE,
    approved_at TIMESTAMP WITH TIME ZONE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);
