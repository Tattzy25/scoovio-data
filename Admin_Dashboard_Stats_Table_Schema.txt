CREATE TABLE admin_dashboard_stats (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    date DATE,
    total_users INT,
    active_users INT,
    total_partners INT,
    active_partners INT,
    total_equipment_listings INT,
    active_equipment_listings INT,
    total_bookings INT,
    completed_bookings INT,
    pending_bookings INT,
    total_revenue DECIMAL(10, 2),
    new_signups_users INT,
    new_signups_partners INT,
    equipment_added_count INT,
    bookings_created_count INT,
    support_tickets_count INT,
    resolved_support_tickets_count INT,
    average_rating DECIMAL(3, 2),
    website_traffic INT,
    mobile_app_downloads INT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);
